cmake_minimum_required(VERSION 3.10)

set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_LIST_DIR}/cmake)
find_package(Tappas REQUIRED)
find_package(HailoRT REQUIRED)
# Find Libcamera
pkg_check_modules(LIBCAMERA REQUIRED libcamera)
message(STATUS "libcamera library found:")
message(STATUS "    version: ${LIBCAMERA_VERSION}")
message(STATUS "    libraries: ${LIBCAMERA_LIBRARIES}")
message(STATUS "    include path: ${LIBCAMERA_INCLUDE_DIRS}")

# Generate a header file with the location of the hailo postprocessing libraries.
file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/hailo_postproc_lib.h"
        "// This file is auto-generated by CMake.\n"
        "#ifndef HAILO_POSTPROC_LIB_DIR_H\n"
        "#define HAILO_POSTPROC_LIB_DIR_H\n"
        "#define HAILO_POSTPROC_LIB_DIR \"${TAPPAS_POSTPROC_LIB_DIR}\"\n"
        "#endif // HAILO_POSTPROC_LIB_DIR_H\n"
)

file(GLOB HAILOPP_CONFIG_FILES
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/yolov5_personface.json"
)

set(HAILO_POSTPROCESSING_SRC
        "${CMAKE_CURRENT_SOURCE_DIR}/hailo_postprocessing_stage.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/hailo_yolo_inference.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/hailo_classifier.cpp"
)

add_library(hailo-postproc SHARED ${HAILO_POSTPROCESSING_SRC})
target_include_directories(hailo-postproc PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/.."
        ${HAILORT_INCLUDE_DIR} ${TAPPAS_INCLUDE_DIR})
target_link_libraries(hailo-postproc PUBLIC ${HAILORT_LIBRARY} ${LIBCAMERA_LIBRARY})
target_compile_options(hailo-postproc PRIVATE ${HAILO_CPP_ARGUMENTS})
install(TARGETS hailo-postproc DESTINATION lib/${PROJECT_NAME}/post_processing_stages)

install(FILES ${HAILOPP_CONFIG_FILES} DESTINATION share/${PROJECT_NAME}/hailo-models)

if (DOWNLOAD_HAILO_MODELS)
    add_custom_target(hailo-models
            COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/utils/download-hailo-models.sh" "${CMAKE_CURRENT_BINARY_DIR}"
    )
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/hailo-models" DESTINATION lib/${PROJECT_NAME})
endif ()
